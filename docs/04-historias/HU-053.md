# HU-053: Crear adaptador OmniRoute

## Resumen
OmniRoute está corriendo en Docker pero no tiene adaptador implementado. Crear adaptador compatible con OpenAI API.

## Descripción Técnica

Como **desarrollador** quiero **crear un adaptador que hable con OmniRoute /v1/chat/completions** para que **el Gateway pueda usar OmniRoute como proveedor**.

## Acceptance Criteria

```gherkin
Given: Adaptador OmniRoute implementado
When: adapter.NewOmniRoute("http://omniroute:20128/v1", apiKey).Chat(ctx, req)
Then: HTTP POST a http://omniroute:20128/v1/chat/completions
And: request body es JSON OpenAI-compatible
And: response parseado correctamente en adapter.Response
And: latencia < 1s

And: error handling:
  - Si OmniRoute no responde: retorn ProviderError con Status 503
  - Si API key inválida: retorna Status 401
  - Si modelo no existe: retorna Status 404
```

## Implementación

### Crear archivo
`src/internal/adapter/omniroute/omniroute.go`

```go
package omniroute

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"time"

	"api-llm-gateway/internal/adapter"
)

type OmniRoute struct {
	baseURL string
	apiKey  string
	client  *http.Client
}

func New(baseURL, apiKey string) *OmniRoute {
	return &OmniRoute{
		baseURL: baseURL,
		apiKey:  apiKey,
		client: &http.Client{
			Timeout: 30 * time.Second,
		},
	}
}

func (o *OmniRoute) Chat(ctx context.Context, req adapter.Request) (adapter.Response, error) {
	// Build request body compatible with OmniRoute OpenAI API
	payload := map[string]interface{}{
		"model":      req.Model,
		"messages":   req.Messages,
		"max_tokens": req.MaxTokens,
	}

	body, err := json.Marshal(payload)
	if err != nil {
		return adapter.Response{}, err
	}

	// POST to OmniRoute
	httpReq, err := http.NewRequestWithContext(ctx, "POST", fmt.Sprintf("%s/chat/completions", o.baseURL), bytes.NewReader(body))
	if err != nil {
		return adapter.Response{}, err
	}

	httpReq.Header.Set("Content-Type", "application/json")
	if o.apiKey != "" {
		httpReq.Header.Set("Authorization", fmt.Sprintf("Bearer %s", o.apiKey))
	}

	resp, err := o.client.Do(httpReq)
	if err != nil {
		return adapter.Response{}, &adapter.ProviderError{
			Provider:  "omniroute",
			Status:    503,
			Retryable: true,
			Err:       err,
		}
	}
	defer resp.Body.Close()

	if resp.StatusCode != 200 {
		respBody, _ := io.ReadAll(resp.Body)
		return adapter.Response{}, &adapter.ProviderError{
			Provider:  "omniroute",
			Status:    resp.StatusCode,
			Retryable: resp.StatusCode >= 500,
			Err:       fmt.Errorf("omniroute: %s", string(respBody)),
		}
	}

	// Parse response
	var result struct {
		Choices []struct {
			Message struct {
				Content string `json:"content"`
			} `json:"message"`
		} `json:"choices"`
	}

	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return adapter.Response{}, err
	}

	if len(result.Choices) == 0 {
		return adapter.Response{}, fmt.Errorf("omniroute: no choices in response")
	}

	return adapter.Response{
		Content: result.Choices[0].Message.Content,
	}, nil
}

func (o *OmniRoute) Embedding(ctx context.Context, req adapter.Request) (adapter.Embedding, error) {
	// TODO: Implement in Fase 2
	return adapter.Embedding{}, fmt.Errorf("omniroute: embeddings not implemented")
}
```
